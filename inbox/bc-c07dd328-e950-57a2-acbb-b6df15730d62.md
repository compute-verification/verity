```lean
namespace VStar.In
/-- One table row of a port as `inst.bin` carries it: its u16 words and its 192-byte salt. -/
structure Row where
  words : Array UInt16
  salt  : ByteArray
/-- A port's input (`circuit.write`, share_rows): its table rows, and the row each instance reads (`refs`). -/
structure PortIn where
  rows : Array Row
  refs : Array Nat
/-- A part's port inputs by name, in its circuit's port order: input ports, then the output rows (`out`, `acc_out`). -/
abbrev PartIn := Array (String × PortIn)

/-- One inner Ligerito opening (`rec_vstar.Query`): its row's elements, siblings bottom up, position. -/
structure Opening where
  rep : Nat
  level : Nat
  j : Nat
  pos : Nat
  row : Array Flock.F128
  sibs : Array ByteArray
/-- What `rec_vstage.stage` reads, decoded (prover's side and verifier's). -/
structure Src where
  program  : String                                -- rec_outer.session_program of public.json
  msgs     : Array (Array Row)                     -- round i, rep r: the inner prover's row and salt (rec-c<i>)
  openings : Array Opening                         -- both reps' (V.queries)
  tops     : Array (Array (Array (ByteArray × ByteArray)))  -- rep, level: each node (64 B) with its firewall salt
  accSalts : Array ByteArray                       -- the N + 1 running sums' salts (--salts)
  kappa    : Array (Array (Array Flock.F128))      -- the final check's κ[level][r][l] at each rep's coins, by rep
  eqA      : Array (Array (Array Flock.F128))      -- eq(α) by rep, level (verifier_rows' info["queries"][level].2)
  vs       : Array (Array Flock.F128)              -- each rep's v (verifier_rows, comb included)
  fresh    : String → Nat → Nat → ByteArray        -- the prover's fresh salt of (dir, port, row): L<l>'s row/aux
/-- Part `st`'s port inputs (`L<l>`: rec_outer.stage; `alg/p<j>`: rec_residuals.write; `cap`, `cap/L<l>`: rec_cap.stage). -/
def partIn (src : Src) : VStar.Stmt → Except String PartIn
-- plus: levelIn src l, algebraIn src j p, capIn src l N, which partIn dispatches to; PortIn.bits (row words as bits, LSB first)
```
